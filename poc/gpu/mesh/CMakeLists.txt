# Mesh workstream targets (poc/gpu/mesh). CPU-only so far. Needs the pinned files from oracle/fetch.sh and the vendored Boost headers.
# Third-party source is fetched into poc/gpu/build/third_party and never committed.
set(MESH_BOOST ${CMAKE_SOURCE_DIR}/../../src/third_party/boost)
set(PRED ${CMAKE_SOURCE_DIR}/build/third_party/shewchuk/predicates.c)
file(GLOB MESH_BOOST_INCS ${MESH_BOOST}/libs/*/include)
if(EXISTS ${PRED} AND EXISTS ${MESH_BOOST}/libs/multiprecision/include)
  add_library(shewchuk_predicates STATIC ${PRED})
  set_source_files_properties(${PRED} PROPERTIES LANGUAGE C)
  # no FMA contraction, no fast math: the predicates rely on exact IEEE operations
  target_compile_options(shewchuk_predicates PRIVATE -ffp-contract=off -w)
  add_executable(predicates_check oracle/predicates_check.cpp)
  target_include_directories(predicates_check PRIVATE ${MESH_BOOST} ${MESH_BOOST_INCS})
  target_compile_options(predicates_check PRIVATE -ffp-contract=off)
  target_link_libraries(predicates_check shewchuk_predicates)
else()
  message(STATUS "mesh: run poc/gpu/mesh/oracle/fetch.sh to enable the predicate oracle targets")
endif()

# MP01 corpus tools (need only the vendored Boost multiprecision headers). -ffp-contract=off keeps the generated bytes identical on every platform.
if(EXISTS ${MESH_BOOST}/libs/multiprecision/include)
  foreach(T gen_corpus corpus_check)
    add_executable(${T} corpus/${T}.cpp)
    target_include_directories(${T} PRIVATE ${MESH_BOOST} ${MESH_BOOST_INCS})
    target_compile_options(${T} PRIVATE -ffp-contract=off)
  endforeach()
endif()
